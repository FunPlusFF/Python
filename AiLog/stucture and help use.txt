├── main.py              # Точка входа с параметрами
├── config.py            # Конфигурация
├── log_reader.py        # Чтение логов из папки
├── parser.py            # Парсинг CSV логов

└── output.py            # Форматирование вывода

# Базовое использование - анализ логов из папки
python main.py /home/gpadmin/gpAdminLogs

# Только FATAL и PANIC ошибки
python main.py /var/log/greenplum -s FATAL

# Сохранение в JSON файл
python main.py /data/logs -o report.json -f json

# Сохранение текстового отчета
python main.py /data/logs -o report.txt -f text

# Фильтрация по времени
python main.py /data/logs --from "2024-01-15 14:00:00" --to "2024-01-15 15:00:00"

# Указание своих паттернов для поиска файлов
python main.py /data/logs --master-pattern "postgres*.csv" --segment-pattern "seg*.csv"

# Просмотр всех уровней (включая DEBUG и LOG)
python main.py /data/logs -s DEBUG
